JavaScript
import { defineLyricsEditor } from 'lyricue';

defineLyricsEditor();
const editor = document.createElement('lyrics-editor');

editor.load({
  audioSrc: '/song.mp3',
  lyrics: 'First line\nSecond line',
  mode: 'tap',
});

editor.addEventListener('lyrics-save', ({ detail }) => {
  console.log(detail.lyrics); // Your LRC, your storage.
});

document.querySelector('#app').append(editor);
